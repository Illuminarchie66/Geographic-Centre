    public centre2(point1, point2)

        lat1 = radians(point1[0])
        lng1 = radians(point1[1])
        lat2 = radians(point2[0])
        lng2 = radians(point2[1])

        Bx = cos(lat2)*cos(lng2-lng1)   
        By = cos(lat2)*sin(lng2-lng1)   

        midlat = atan2( sin(lat1) + sin(lat2), sqrt((cos(lat1) + Bx)**2 +By**2))   
        midlng = lng1 + atan2(By, cos(lat1) + Bx)

        midpoint = Point.fromPolar([midlat, midlng])

        return midpoint
